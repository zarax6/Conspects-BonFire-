### 1
``` SQL
SELECT model
FROM (
    SELECT model, price
    FROM PC
    UNION ALL
    SELECT model, price
    FROM Laptop
    UNION ALL
    SELECT model, price
    FROM Printer
) AS Products
GROUP BY model
HAVING MAX(price) >= ALL (
    SELECT price FROM PC
    UNION ALL
    SELECT price FROM Laptop
    UNION ALL
    SELECT price FROM Printer
);
```

### 2
``` SQL
SELECT p.maker
FROM Productt p
JOIN PC pc ON p.model = pc.model
GROUP BY p.maker
HAVING MIN(pc.ram) = (
    SELECT MIN(ram)
    FROM PC
)
AND MAX(pc.speed) >= ALL (
    SELECT speed
    FROM PC
    WHERE ram = (
        SELECT MIN(ram)
        FROM PC
    )
)
AND p.maker IN (
    SELECT maker
    FROM Productt
    WHERE type = 'Printer'
);
```
