### 9
``` SQL
UPDATE Laptop
SET hd = hd * 1.5,
    price = price * 1.07
WHERE model IN (
    SELECT model
    FROM Productt
    WHERE maker = 'C'
)
AND screen < 16
AND price <= 1500;
```

``` SQL
UPDATE PC
SET price = CASE
    WHEN ram >= 4096 THEN price * 0.90
    WHEN ram BETWEEN 1024 AND 4095 THEN price
    WHEN ram < 1024 THEN price * 1.05
END;
```
