### 3
``` SQL
SELECT
    B.name AS BattleName,
    B.date AS BattleDate,
    S.name AS ShipName,
    S.class AS ClassName,
    O.result AS Result
FROM Ships S
INNER JOIN Outcomes O
    ON S.name = O.ship
INNER JOIN Battles B
    ON O.battle = B.name
WHERE B.date IS NOT NULL
ORDER BY
    B.date,
    S.name;
```

``` SQL
SELECT
    country,
    COUNT(*) AS ClassCount
FROM Classes
GROUP BY country
HAVING COUNT(*) >= 2
ORDER BY
    ClassCount DESC,
    country ASC;
```

### 7
``` SQL
CREATE TABLE Captains (
    captain_id INT NOT NULL,
    full_name VARCHAR(100) NOT NULL,
    [rank] VARCHAR(20) NULL,
    country VARCHAR(50) NOT NULL,
    birth_year INT NULL,
    ship_name VARCHAR(50) NULL,

    CONSTRAINT PK_Captains
        PRIMARY KEY (captain_id),

    CONSTRAINT UQ_Captains_FullName
        UNIQUE (full_name),

    CONSTRAINT UQ_Captains_ShipName
        UNIQUE (ship_name),

    CONSTRAINT CK_Captains_Rank
        CHECK ([rank] IN (
            'Lieutenant',
            'Commander',
            'Captain',
            'Admiral'
        ) OR [rank] IS NULL),

    CONSTRAINT CK_Captains_BirthYear
        CHECK (birth_year BETWEEN 1880 AND 1925),

    CONSTRAINT FK_Captains_Ships
        FOREIGN KEY (ship_name)
        REFERENCES Ships(name)
);
GO
```
