##  2
Условный оператор if. Синтаксис, Вложенные условия и блок else

##### if 
условие, имеет два состояния (типа bool) true и false.

Как выглядит, после if идут скобки в которых и пишем условие, дальше идут {} в них мы пишем само "Тело" что будем делать если условие True к нашему условию.

Дальше идет блок else, который выполняет какое то действие если в if значение не приравнялось к true (Не подходило)

## 3
Понятие одномерного массива. Объявление, выделение памяти и инициализация.

##### Массив
одномерный массив - Линейный (Строчный) список элементов, у каждого элемента ровно 1 индекс, не 2 как в двумерном. Также в C# Массивы - ссылочный тип.

##### Объявление
Сперва мы создаем переменную ссылку (int[] numb;)

Дальше мы выделяем память обращаюсь по ссылке давая количество элементов через оператор (new), После выделения у нас будут все элементы нулями.

Пример: int - 0, double - 0.0, bool - false, char - '\0', string - null

Обычно объявление и выделение объединяют (int[] numbers = new int[5])

##### Инициализация

через new только с длиной 
int[] = new int[3]; будет 0, 0, 0

Через new со списком значений 

int b[] = new int[] {10, 12, 14}

и обычная (Сокращенная)

int b[3] = {10, 20, 30}

``` C#
namespace Билет_Максим_Северюхин
{
	internal class Program
	{
		static void Main(string[] args)
		{
			Console.WriteLine("a = "); double a = double.Parse(Console.ReadLine());
			Console.WriteLine("b = "); double b = double.Parse(Console.ReadLine());
			Console.WriteLine("c = "); double c = double.Parse(Console.ReadLine());

			if (a == 0)
			{
				if (b != 0) Console.WriteLine($"Корень x = {-c / b}");
				else Console.WriteLine("Корней нет или их бесконечно много");
				return;
			}

			double d = b * b - 4 * a * c;

			if (d > 0)
			{
				Console.WriteLine($"Два корня: x1 = {(-b + Math.Sqrt(d)) / (2 * a)}, x2 = {(-b - Math.Sqrt(d)) / (2 * a)}");
			}
			else if (d == 0)
			{
				Console.WriteLine($"Один корень: x = {-b / (2 * a)}");
			}
			else
			{
				Console.WriteLine("Нет действительных корней");
			}
		}
	}
}
```

## 2
``` C#
using System.Diagnostics.CodeAnalysis;

namespace Билет_Максим_Северюхин_2
{
    internal class Program
    {
        static void Main(string[] args)
        {
            Console.WriteLine("Введите целое положительное число: ");
            int number = int.Parse(Console.ReadLine());

            int sum = 0;

            while (number > 0)
            {
                sum += number % 10;
                number /= 10;
            }
            Console.WriteLine($"Сумма цифр: {sum}");
            Console.ReadKey();
        }
    }
}
```

``` C#
namespace Билет_МС_3
{
    internal class Program
    {
        static void Main(string[] args)
        {
            int[] arr = { -1, 23, 2, 5, -21, 4, 6, -12, 4, 15, -11 };
            int sum = 0;

            foreach (int x in arr)
            {
                if (x < 0) sum += x;
            }

            Console.WriteLine(" Сумма отриц чисел = "+ sum);
        }
    }
}
```
