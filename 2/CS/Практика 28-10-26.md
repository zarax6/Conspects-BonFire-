``` C#
using System.Net.NetworkInformation;

namespace Практика_28_10_26
{
    using System;

    class Student
    {
        public string Name = "";
        public int Score;
    }

    class Program
    {
        static void Main()
        {
            Student[] group =
            {
            new Student { Name = "Анна", Score = 5 },
            new Student { Name = "Илья", Score = 4 },
            new Student { Name = "Мария", Score = 3 },
            new Student { Name = "Олег", Score = 5 },
            new Student { Name = "Анна", Score = 4 },
            new Student { Name = "Денис", Score = 3 }
        };

            int sum = 0;
            int max = group[0].Score;
            int count3 = 0;
            int count4 = 0;
            int count5 = 0;

            Console.WriteLine("Журнал группы");
            Console.WriteLine();

            for (int i = 0; i < group.Length; i++)
            {
                Console.WriteLine($"{i + 1}. {group[i].Name} - {group[i].Score}");

                sum += group[i].Score;

                if (group[i].Score > max)
                    max = group[i].Score;

                if (group[i].Score == 3)
                    count3++;
                if (group[i].Score == 4)
                    count4++;
                if (group[i].Score == 5)
                    count5++;
            }

            double average = (double)sum / group.Length;

            Console.WriteLine();
            Console.WriteLine("Максимальная оценка:");

            for (int i = 0; i < group.Length; i++)
            {
                if (group[i].Score == max)
                    Console.WriteLine($"{group[i].Name} - {group[i].Score}");
            }

            Console.WriteLine($"Средний балл: {average:F2}");
            Console.WriteLine($"Оценок 3: {count3}");
            Console.WriteLine($"Оценок 4: {count4}");
            Console.WriteLine($"Оценок 5: {count5}");

            Console.WriteLine();
            Console.WriteLine("Обратный порядок:");

            for (int i = group.Length - 1; i >= 0; i--)
            {
                Console.WriteLine($"{group[i].Name} - {group[i].Score}");
            }

            Console.WriteLine();
            Console.WriteLine("Студенты с оценкой ниже среднего:");

            for (int i = 0; i < group.Length; i++)
            {
                if (group[i].Score < average)
                    Console.WriteLine($"{group[i].Name} - {group[i].Score}");
            }

            Console.WriteLine();
            Console.Write("Введите имя для поиска: ");
            string name = Console.ReadLine() ?? "";
            bool found = false;

            for (int i = 0; i < group.Length; i++)
            {
                if (group[i].Name.Equals(name, StringComparison.OrdinalIgnoreCase))
                {
                    Console.WriteLine($"{group[i].Name} - {group[i].Score}");
                    found = true;
                }
            }

            if (!found)
                Console.WriteLine("Студенты с таким именем не найдены.");
        }
    }
}
```

### Ответы на контрольные вопросы

1. Чем класс отличается от объекта? Класс описывает структуру и данные, а объект является конкретным экземпляром этого класса.
    
2. Что означает выражение group[i].Name? Оно обращается к имени студента, который находится в массиве group под индексом i.
    
3. Для чего используется свойство Length? Оно возвращает количество элементов массива.
    
4. Почему удобно хранить однотипные записи в массиве объектов? Потому что можно хранить много студентов в одной переменной и обрабатывать их циклами, не создавая отдельную переменную для каждого студента.

``` C#
namespace Практичская_2_28_10_26
{

    using System;

    class Program
    {
        class Student
        {
            public string Name = "";
            public int Score;
        }

        static void Main()
        {
            Student studentA = new Student { Name = "Анна", Score = 5 };
            Student studentB = studentA;

            studentB.Score = 2;
            Console.WriteLine("Оценка через studentA: " + studentA.Score);

            studentA.Name = "Мария";
            Console.WriteLine("Имя через studentB: " + studentB.Name);

            Student studentC = new Student { Name = "Илья", Score = 4 };
            studentC.Score = 3;
            Console.WriteLine("Оценка studentA: " + studentA.Score);
            Console.WriteLine("Оценка studentC: " + studentC.Score);

            Console.WriteLine();

            Student[] group =
            {
            new Student { Name = "Анна", Score = 5 },
            new Student { Name = "Илья", Score = 4 },
            new Student { Name = "Мария", Score = 3 },
            new Student { Name = "Олег", Score = 5 }
        };

            Student[] groupCopy = group;

            groupCopy[0] = new Student { Name = "Пётр", Score = 2 };
            Console.WriteLine("group[0]: " + group[0].Name);

            groupCopy[1].Score = 2;
            Console.WriteLine("Оценка group[1]: " + group[1].Score);

            Console.WriteLine();

            Student[] groupClone = (Student[])group.Clone();

            groupClone[0] = new Student { Name = "Денис", Score = 3 };
            Console.WriteLine("group[0] после Clone: " + group[0].Name);

            groupClone[1].Score = 5;
            Console.WriteLine("Оценка group[1] после изменения clone: " + group[1].Score);

            Console.WriteLine();

            Student[] students = new Student[3];

            if (students[0] == null)
                Console.WriteLine("Первый элемент равен null");

            students[0] = new Student { Name = "Елена", Score = 4 };
            Console.WriteLine(students[0].Name + " - " + students[0].Score);
        }
    }

}
```

### Краткие ответы на вопросы

1. Что копируется при присваивании объекта? Копируется ссылка на объект. Обе переменные обращаются к одному объекту, поэтому изменение его полей видно через обе переменные.
    
2. Что копируется при присваивании массива? Копируется ссылка на массив. Обе переменные указывают на один и тот же массив.
    
3. Чем поверхностная копия массива отличается от независимых копий объектов? Метод Clone() создаёт новый массив, но ссылки на объекты внутри него остаются общими. Поэтому замена элемента не меняет исходный массив, а изменение поля общего объекта видно через оба массива.
    
4. Зачем проверять ссылку на null? Чтобы убедиться, что объект существует, прежде чем обращаться к его полям. Иначе программа может завершиться с ошибкой NullReferenceException.
