### Код
``` C#
namespace ДЗ_21_09
{
    internal class Program
    {
        static void Main(string[] args)
        {
            Console.Write("Введите количество элементов: ");
            int n = int.Parse(Console.ReadLine());
            
            double[] a = new double[n];
            
            for (int i = 0; i < n; i++)
            {
                Console.Write("a[" + i + "] = ");
                a[i] = double.Parse(Console.ReadLine());
            }
            
            double sumNeg = 0;
            
            for (int i = 0; i < n; i++)
            {
                if (a[i] < 0)
                    sumNeg += a[i];
            }
            
            int min = 0, max = 0;
            
            for (int i = 1; i < n; i++)
            {
                if (a[i] < a[min]) min = i;
                if (a[i] > a[max]) max = i;
            }
            
            double prodMinMax = 1;
            
            int start = Math.Min(min, max);
            int end = Math.Max(min, max);
            
            for (int i = start + 1; i < end; i++)
                prodMinMax *= a[i];
                
            double prodEven = 1;
            for (int i = 1; i < n; i += 2)
                prodEven *= a[i];
            int first = -1, last = -1;
            for (int i = 0; i < n; i++)
            {
                if (a[i] < 0)
                {
                    if (first == -1)
                        first = i;
                    last = i;
                }
            }
            double sumBetween = 0;
            if (first != -1 && last != -1)
            {
                for (int i = first + 1; i < last; i++)
                    sumBetween += a[i];
            }
            Console.WriteLine("\nРезультаты:");
            Console.WriteLine("1. Сумма отрицательных элементов: " + sumNeg);
            Console.WriteLine("2. Произведение между min и max: " + prodMinMax);
            Console.WriteLine("3. Произведение элементов с четными номерами: " + prodEven);
            Console.WriteLine("4. Сумма между первым и последним отрицательными: " + sumBetween);
        }
    }
}
```
